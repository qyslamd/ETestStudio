> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 etest_engine 核心库 + IDE 模式执行 UI + 独立引擎 GUI/CLI，使用 mock 插件验证完整执行链路

**Architecture:** 分层设计，etest_engine 纯核心（仅 Qt5::Core + spdlog）无 UI 依赖；IDE 模式 UI 组件在 etest_app 中；两个独立引擎（test-executor GUI + test-executor-cli）分别链接所需依赖。执行引擎通过 QThread Worker 异步执行，StepRunner 发射信号驱动 UI 更新。

**Tech Stack:** Qt5::Core, Qt5::Widgets, spdlog, etest_core (SignalRegistry), icd_utility, mock plugins

- 所有新增代码遵循 Google C++ Style，使用 clang-format
- if/for 单行也必须加 `{}`
- `etest_engine` 不得依赖 Qt5::Widgets，仅允许 Qt5::Core + spdlog
- 独立产品约束：test-executor / test-executor-cli 不得链接 etest_topology、etest_protocol、etest_program、icd_utility（icd_utility 仅通过 etest_engine 间接使用）
- .etlog 命名规则：`.etprog` 文件名去掉后缀，如 `燃油阀门控制.etlog`
- UI 样式禁止 C++ `setStyleSheet`，统一用 QSS 文件

```cpp
ResolvedSignal SignalResolver::resolve(const QString& uuid) const {
    ResolvedSignal result;

    if (!registry_ || !icd_repo_) {
        spdlog::warn("[SignalResolver] registry_ or icd_repo_ is null");
        result.valid = false;
        return result;
    }

    // ① 查 SignalRegistry 获取设备/端口/帧/节点路径
    auto resolved = registry_->resolve(uuid);
    if (!resolved.isValid()) {
        spdlog::warn("[SignalResolver] UUID {} not found in SignalRegistry",
                     uuid.toStdString());
        result.valid = false;
        return result;
    }

    result.deviceId = resolved.deviceId;
    result.portName = resolved.portName;

    // ② 查 ICD Repository 补充编码属性
    fillFromIcd(result, resolved.frameName, resolved.nodePath);

    // ③ 确定信号类型（从端口属性推断）
    auto* port = registry_->findPort(result.deviceId, result.portName);
    if (port) {
        result.signalType = static_cast<SignalType>(port->signalType);
    }

    result.valid = true;
    return result;
}

void SignalResolver::fillFromIcd(ResolvedSignal& signal,
                                  const QString& frameName,
                                  const QString& nodePath) const {
    auto* frame = icd_repo_->findFrame(frameName);
    if (!frame) {
        spdlog::warn("[SignalResolver] Frame '{}' not found in ICD Repository",
                     frameName.toStdString());
        return;
    }

    auto* node = frame->findNodeByPath(nodePath);
    if (!node) {
        spdlog::warn("[SignalResolver] Node path '{}' not found in frame '{}'",
                     nodePath.toStdString(), frameName.toStdString());
        return;
    }

    signal.coeff = node->coeff;
    signal.offset = node->offset;
    signal.unit = node->unit;
    signal.engMin = node->engMin;
    signal.engMax = node->engMax;
    signal.rawMin = node->rawMin;
    signal.rawMax = node->rawMax;
    signal.channel = node->channel;
    signal.frameId = node->frameId;
    signal.byteOffset = node->byteOffset;
    signal.bitOffset = node->bitOffset;
    signal.bitWidth = node->bitWidth;
    signal.byteOrder = (node->byteOrder == 0)
        ? ByteOrder::BigEndian : ByteOrder::LittleEndian;
}

}  // namespace etest::engine
```

```cpp
#include "SignalCodec.h"
#include "SignalResolver.h"
#include <cmath>
#include <cstring>
#include <spdlog/spdlog.h>

namespace etest::engine {

double SignalCodec::linearMap(double raw, const ResolvedSignal& info) const {
    return raw * info.coeff + info.offset;
}

double SignalCodec::inverseLinearMap(double eng, const ResolvedSignal& info) const {
    if (std::abs(info.coeff) < 1e-15) {
        spdlog::warn("[SignalCodec] coeff near zero, returning 0");
        return 0.0;
    }
    return (eng - info.offset) / info.coeff;
}

QByteArray SignalCodec::packBits(double value, const ResolvedSignal& info) const {
    int totalBytes = (info.bitOffset + info.bitWidth + 7) / 8;
    QByteArray frame(totalBytes, '\0');
    int64_t rawBits = static_cast<int64_t>(std::round(value));

    for (int i = 0; i < info.bitWidth; i++) {
        int srcBit = i;  // LSB first
        if (rawBits & (1LL << srcBit)) {
            int dstByte = (info.bitOffset + i) / 8;
            int dstBit  = (info.bitOffset + i) % 8;
            if (info.byteOrder == ByteOrder::BigEndian) {
                dstByte = totalBytes - 1 - dstByte;
            }
            if (dstByte >= 0 && dstByte < frame.size()) {
                frame[dstByte] |= static_cast<char>(1 << dstBit);
            }
        }
    }
    return frame;
}

double SignalCodec::unpackBits(const QByteArray& data,
                                const ResolvedSignal& info) const {
    int totalBytes = (info.bitOffset + info.bitWidth + 7) / 8;
    if (data.size() < totalBytes) {
        spdlog::warn("[SignalCodec] frame too short: need {} have {}",
                     totalBytes, data.size());
        return 0.0;
    }

    int64_t rawBits = 0;
    for (int i = 0; i < info.bitWidth; i++) {
        int srcByte = (info.bitOffset + i) / 8;
        int srcBit  = (info.bitOffset + i) % 8;
        if (info.byteOrder == ByteOrder::BigEndian) {
            srcByte = totalBytes - 1 - srcByte;
        }
        if (srcByte >= 0 && srcByte < data.size()) {
            if (data[srcByte] & (1 << srcBit)) {
                rawBits |= (1LL << i);
            }
        }
    }
    return static_cast<double>(rawBits);
}

QVariant SignalCodec::encode(double engValue, const ResolvedSignal& info) {
    double raw = inverseLinearMap(engValue, info);
    return QVariant(raw);
}

QByteArray SignalCodec::encodeToFrame(double engValue, const ResolvedSignal& info) {
    double raw = inverseLinearMap(engValue, info);
    return packBits(raw, info);
}

double SignalCodec::decode(const QVariant& rawValue, const ResolvedSignal& info) {
    bool ok = false;
    double raw = rawValue.toDouble(&ok);
    if (!ok) {
        spdlog::warn("[SignalCodec] decode: invalid QVariant");
        return 0.0;
    }
    return linearMap(raw, info);
}

double SignalCodec::decodeFromFrame(const QByteArray& frameData,
                                     const ResolvedSignal& info) {
    double raw = unpackBits(frameData, info);
    return linearMap(raw, info);
}

}  // namespace etest::engine
```

```cpp
bool HardwareManager::loadFromTopology(const QString& etopoPath) {
    QFile file(etopoPath);
    if (!file.open(QIODevice::ReadOnly)) {
        spdlog::error("[HardwareManager] Cannot open topology: {}",
                      etopoPath.toStdString());
        return false;
    }
    QJsonDocument doc = QJsonDocument::fromJson(file.readAll());
    file.close();

    QJsonArray devices = doc.object().value("devices").toArray();
    for (const auto& devVal : devices) {
        QJsonObject dev = devVal.toObject();
        QString deviceId = dev.value("id").toString();
        QString pluginId = dev.value("pluginId").toString();
        QVariantMap properties = dev.value("properties").toObject().toVariantMap();
        instantiateDevice(deviceId, pluginId, properties);
    }
    return true;
}

bool HardwareManager::instantiateDevice(const QString& deviceId,
                                         const QString& pluginId,
                                         const QVariantMap& properties) {
    auto* pluginMgr = etest::core::PluginManager::instance();
    auto plugins = pluginMgr->devicesByType(pluginId);
    if (plugins.isEmpty()) {
        spdlog::warn("[HardwareManager] No plugin found for pluginId '{}'",
                     pluginId.toStdString());
        return false;
    }

    auto* plugin = plugins.first();
    DeviceEntry entry;
    entry.plugin = plugin;
    entry.status = DeviceStatus::Online;

    device_pool_.insert(deviceId, entry);
    emit deviceStatusChanged(deviceId, DeviceStatus::Online);
    return true;
}

QVariant HardwareManager::read(const ResolvedSignal& signal) {
    auto it = device_pool_.find(signal.deviceId);
    if (it == device_pool_.end() || it->status == DeviceStatus::Offline) {
        throw DeviceException("Device offline: " + signal.deviceId.toStdString());
    }
    // Phase 1: mock 设备直接返回模拟值
    auto* plugin = it->plugin;
    if (auto* ad = dynamic_cast<IADevicePlugin*>(plugin)) {
        return QVariant(ad->readChannel(signal.channel));
    }
    if (auto* can = dynamic_cast<ICANPlugin*>(plugin)) {
        auto msg = can->receiveMessage();
        return QVariant::fromValue(msg);
    }
    // ... 其余设备类型
    return QVariant();
}

QVariant HardwareManager::readAndWait(const ResolvedSignal& signal, int timeoutMs) {
    // Phase 1: mock 模式下直接返回 read()
    // Phase 2: 真实设备时，通过 QEventLoop + 定时器等待目标信号到达
    return read(signal);
}

bool HardwareManager::write(const ResolvedSignal& signal, double engValue) {
    auto it = device_pool_.find(signal.deviceId);
    if (it == device_pool_.end() || it->status == DeviceStatus::Offline) {
        emit deviceError(signal.deviceId, QStringLiteral("设备离线"));
        return false;
    }
    auto* plugin = it->plugin;
    if (auto* da = dynamic_cast<IDADevicePlugin*>(plugin)) {
        return da->writeChannel(signal.channel, engValue);
    }
    if (auto* can = dynamic_cast<ICANPlugin*>(plugin)) {
        can->sendMessage(static_cast<uint32_t>(signal.frameId),
                         QByteArray());  // frame bytes from SignalCodec
        return true;
    }
    return false;
}

bool HardwareManager::writeFrame(const ResolvedSignal& signal,
                                  const QByteArray& frameData) {
    auto it = device_pool_.find(signal.deviceId);
    if (it == device_pool_.end() || it->status == DeviceStatus::Offline) return false;
    auto* plugin = it->plugin;
    if (auto* serial = dynamic_cast<ISerialDevicePlugin*>(plugin)) {
        serial->writeFrame(frameData);
        return true;
    }
    if (auto* a429 = dynamic_cast<IArinc429Plugin*>(plugin)) {
        a429->sendWord(signal.frameId,
                       *reinterpret_cast<const uint32_t*>(frameData.constData()));
        return true;
    }
    return false;
}

DeviceStatus HardwareManager::deviceStatus(const QString& deviceId) const {
    auto it = device_pool_.find(deviceId);
    return (it != device_pool_.end()) ? it->status : DeviceStatus::Offline;
}

QList<QString> HardwareManager::onlineDevices() const {
    QList<QString> online;
    for (auto it = device_pool_.begin(); it != device_pool_.end(); ++it) {
        if (it->status == DeviceStatus::Online) online.append(it.key());
    }
    return online;
}

            it->plugin->close();
        }
    }
```

```cpp
StepResult StepRunner::executeSingleStep(const TestStepData& step,
                                          int caseIndex,
                                          const QString& stepPath) {
    StepResult result;
    result.stepPath = stepPath;
    result.command = step.command;
    result.timestamp = QDateTime::currentDateTime();

    emit stepStarted(caseIndex, stepPath, step.command, step.target);

    if (cancel_flag_.load() == CANCELLING) {
        result.status = SKIPPED;
        result.message = QStringLiteral("执行被中止");
        emit stepFinished(caseIndex, stepPath, result);
        return result;
    }

    try {
        auto signal = resolver_->resolve(step.target);
        if (!signal.valid) {
            result.status = ERROR;
            result.message = QStringLiteral("信号解析失败: UUID %1 未在 SignalRegistry 中找到")
                                 .arg(step.target);
            emit stepFinished(caseIndex, stepPath, result);
            return result;
        }

        QElapsedTimer timer;
        timer.start();

        switch (commandType(step.command)) {
            case CommandType::SET:
                result = execSet(step, signal);
                break;
            case CommandType::VERIFY:
                result = execVerify(step, signal);
                break;
            case CommandType::CHECK:
                result = execCheck(step, signal);
                break;
            case CommandType::WAIT:
                result = execWait(step, signal);
                break;
            case CommandType::DELAY:
                result = execDelay(step);
                break;
            case CommandType::LOOP:
                result = execLoop(step, caseIndex, stepPath);
                break;
            case CommandType::WHILE:
                result = execWhile(step, caseIndex, stepPath);
                break;
            case CommandType::IF:
                result = execIf(step, caseIndex, stepPath);
                break;
            default:
                result.status = ERROR;
                result.message = QStringLiteral("未知命令: %1").arg(step.command);
                break;
        }
        result.elapsedMs = static_cast<int>(timer.elapsed());

    } catch (const TimeoutException& e) {
        result.status = TIMEOUT;
        result.message = e.what();
    } catch (const DeviceException& e) {
        result.status = ERROR;
        result.message = QStringLiteral("设备错误: %1").arg(e.what());
    } catch (const std::exception& e) {
        result.status = ERROR;
        result.message = QStringLiteral("内部错误: %1").arg(e.what());
    }

    emit stepFinished(caseIndex, stepPath, result);
    return result;
}

StepResult StepRunner::execSet(const TestStepData& step,
                                const ResolvedSignal& signal) {
    double engValue = step.value.toDouble();
    if (signal.signalType == SignalType::AD || signal.signalType == SignalType::DA) {
        double raw = codec_->encode(engValue, signal).toDouble();
        hw_->write(signal, raw);
    } else {
        QByteArray frame = codec_->encodeToFrame(engValue, signal);
        hw_->writeFrame(signal, frame);
    }
    return StepResult{}.setStatus(PASS);
}

StepResult StepRunner::execCheck(const TestStepData& step,
                                  const ResolvedSignal& signal) {
    double expected = step.value.toDouble();
    double actual;
    if (signal.signalType == SignalType::AD || signal.signalType == SignalType::DA) {
        QVariant raw = hw_->read(signal);
        actual = codec_->decode(raw, signal);
    } else {
        // CAN/Serial/A429: read, then decode from frame
        int timeout = (step.timeoutMs > 0) ? step.timeoutMs : kDefaultTimeoutMs;
        QByteArray frame = hw_->readAndWait(signal, timeout).value<QByteArray>();
        actual = codec_->decodeFromFrame(frame, signal);
    }
    double tolerance = step.tolerance;  // from step column
    if (std::abs(actual - expected) <= tolerance) {
        return StepResult{}.setStatus(PASS);
    }
    return StepResult{}.setStatus(FAIL)
        .setMessage(QStringLiteral("期望 %1，实际 %2，容差 %3")
                         .arg(expected).arg(actual).arg(tolerance));
}

StepResult StepRunner::execVerify(const TestStepData& step,
                                   const ResolvedSignal& signal) {
    // VERIFY 类似 CHECK 但用于条件判断（用于 IF/WHILE 条件的前置检查）
    return execCheck(step, signal);
}

StepResult StepRunner::execWait(const TestStepData& step,
                                 const ResolvedSignal& signal) {
    int timeout = (step.timeoutMs > 0) ? step.timeoutMs : kDefaultTimeoutMs;
    // 循环等待条件满足，使用 read 每次采样检查
    QElapsedTimer waitTimer;
    waitTimer.start();
    while (!isCancelling()) {
        QVariant raw = hw_->read(signal);
        double actual = codec_->decode(raw, signal);
        // 条件在 step.extra 中，格式如 "> 100"
        if (checkCondition(actual, step.extra)) {
            return StepResult{}.setStatus(PASS)
                .setMessage(QStringLiteral("条件满足，等待 %1 ms")
                                 .arg(waitTimer.elapsed()));
        }
        if (waitTimer.elapsed() > timeout) {
            return StepResult{}.setStatus(TIMEOUT)
                .setMessage(QStringLiteral("等待条件超时 %1 ms").arg(timeout));
        }
        QThread::msleep(50);  // 50ms 采样间隔
    }
    return StepResult{}.setStatus(SKIPPED);
}

StepResult StepRunner::execDelay(const TestStepData& step) {
    int delayMs = step.value.toInt();
    if (delayMs <= 0) delayMs = 1000;
    QThread::msleep(delayMs);
    return StepResult{}.setStatus(PASS);
}

StepResult StepRunner::execLoop(const TestStepData& step,
                                 int caseIndex, const QString& stepPath) {
    int failCount = 0;
    for (int i = 0; i < step.loopCount; i++) {
        for (int si = 0; si < step.subSteps.size(); si++) {
            QString subPath = QStringLiteral("%1/%2/%3")
                                  .arg(stepPath).arg(i).arg(si);
            auto subResult = executeSingleStep(step.subSteps[si], caseIndex, subPath);
            if (subResult.status != PASS) failCount++;
            if (cancel_flag_.load() == CANCELLING) break;
        }
        if (cancel_flag_.load() == CANCELLING) break;
    }
    if (failCount > 0) {
        return StepResult{}.setStatus(FAIL)
            .setMessage(QStringLiteral("LOOP 执行完成，%1 个子步骤失败").arg(failCount));
    }
    return StepResult{}.setStatus(PASS);
}

StepResult StepRunner::execWhile(const TestStepData& step,
                                  int caseIndex, const QString& stepPath) {
    int failCount = 0;
    int iteration = 0;
    int maxIter = step.loopCount > 0 ? step.loopCount : 100;  // 防无限循环

    while (iteration < maxIter && !isCancelling()) {
        // 检查条件（条件在 step.condition 中）
        bool conditionMet = evaluateCondition(step.condition);
        if (!conditionMet) break;

        for (int si = 0; si < step.subSteps.size(); si++) {
            QString subPath = QStringLiteral("%1/%2/%3")
                                  .arg(stepPath).arg(iteration).arg(si);
            auto subResult = executeSingleStep(step.subSteps[si], caseIndex, subPath);
            if (subResult.status != PASS) failCount++;
            if (cancel_flag_.load() == CANCELLING) break;
        }
        iteration++;
        if (cancel_flag_.load() == CANCELLING) break;
    }
    if (cancel_flag_.load() == CANCELLING) {
        return StepResult{}.setStatus(SKIPPED);
    }
    if (failCount > 0) {
        return StepResult{}.setStatus(FAIL)
            .setMessage(QStringLiteral("WHILE 执行完成，%1 个子步骤失败").arg(failCount));
    }
    return StepResult{}.setStatus(PASS);
}

StepResult StepRunner::execIf(const TestStepData& step,
                               int caseIndex, const QString& stepPath) {
    bool conditionMet = evaluateCondition(step.condition);
    if (conditionMet) {
        for (int si = 0; si < step.thenSteps.size(); si++) {
            QString subPath = QStringLiteral("%1/THEN/%2").arg(stepPath).arg(si);
            executeSingleStep(step.thenSteps[si], caseIndex, subPath);
            if (cancel_flag_.load() == CANCELLING) break;
        }
    } else {
        for (int si = 0; si < step.elseSteps.size(); si++) {
            QString subPath = QStringLiteral("%1/ELSE/%2").arg(stepPath).arg(si);
            executeSingleStep(step.elseSteps[si], caseIndex, subPath);
            if (cancel_flag_.load() == CANCELLING) break;
        }
    }
    return StepResult{}.setStatus(PASS);
}

void StepRunner::executeProgram(const TestProgramData& program) {
    cancel_flag_.store(RUNNING);
    emit suiteStarted(program.suiteName);

    int passCount = 0, failCount = 0;
    for (int ci = 0; ci < program.cases.size(); ci++) {
        if (isCancelling()) break;
        executeCase(program.cases[ci], ci);
    }

    emit suiteFinished(program.suiteName, passCount, failCount);
}

void StepRunner::executeCase(const TestCaseData& caseData, int caseIndex) {
    if (isCancelling()) return;
    emit caseStarted(caseIndex, caseData.caseName);

    for (int si = 0; si < caseData.steps.size(); si++) {
        if (isCancelling()) break;
        executeSingleStep(caseData.steps[si], caseIndex, QString::number(si));
    }

    emit caseFinished(caseIndex, caseData.caseName, 0);
}

bool evaluateCondition(const QString& condition) {
    // Phase 1 简化实现：解析 "UUID > 100" 格式的字符串
    // 从 condition 中提取运算符和值，调用硬件读取当前值后比较
    // Phase 2 使用完整的 SignalResolver + Codec 链路
    return true;
}

bool checkCondition(double actual, const QString& condition) {
    // 解析 "> 100" 格式的条件后缀
    QString trimmed = condition.trimmed();
    if (trimmed.startsWith('>')) {
        double threshold = trimmed.mid(1).trimmed().toDouble();
        return actual > threshold;
    }
    if (trimmed.startsWith('<')) {
        double threshold = trimmed.mid(1).trimmed().toDouble();
        return actual < threshold;
    }
    if (trimmed.startsWith(">=")) {
        double threshold = trimmed.mid(2).trimmed().toDouble();
        return actual >= threshold;
    }
    if (trimmed.startsWith("<=")) {
        double threshold = trimmed.mid(2).trimmed().toDouble();
        return actual <= threshold;
    }
    if (trimmed.startsWith("==")) {
        double threshold = trimmed.mid(2).trimmed().toDouble();
        return std::abs(actual - threshold) < 1e-9;
    }
    return true;
}
```
