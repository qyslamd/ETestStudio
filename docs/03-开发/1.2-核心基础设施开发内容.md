## 一、日志系统

### 1.1 现有实现

```cpp
// src/core/logger/Logger.h
namespace etest::core::logger {

// 日志级别
enum LogLevel {
  LOG_LEVEL_DEBUG = spdlog::level::debug,
  LOG_LEVEL_INFO = spdlog::level::info,
  LOG_LEVEL_WARN = spdlog::level::warn,
  LOG_LEVEL_ERROR = spdlog::level::err,
  LOG_LEVEL_FATAL = spdlog::level::critical
};

class Logger {
 public:
  static void init();           // 初始化日志系统
  static void shutdown();         // 关闭日志系统
  static void setLevel(const QString& module, LogLevel level);
  static void setAllLevel(LogLevel level);
  static void log(const QString& module, LogLevel level, const char* file, int line, const char* format, ...);
};

}  // namespace etest::core::logger
```

### 1.2 使用方式

```cpp
// 全局日志调用宏
LOG_DEBUG(module, "日志内容 {}", param);
LOG_INFO(module, "日志内容 {}", param);
LOG_WARN(module, "日志内容 {}", param);
LOG_ERROR(module, "日志内容 {}", param);
LOG_FATAL(module, "日志内容 {}", param);

// 示例
LOG_INFO("MAIN", "程序启动，版本: {}", APP_VERSION);
LOG_ERROR("NETWORK", "连接失败: {}", error_string);
```

### 1.3 配置项

| 配置键 | 类型 | 默认值 | 说明 |
|--------|------|--------|------|

| `log/max_file_size` | int | 10485760 | 单个日志文件最大大小（字节） |
| `log/max_file_count` | int | 20 | 保留的日志文件数量 |
| `log/keep_days` | int | 7 | 保留的历史日志天数 |

### 1.4 日志输出

- 控制台：彩色输出，自动按级别着色

### 5.3 GlobalExceptionHandler（新增实现）

```cpp
// src/core/common/GlobalExceptionHandler.h
#ifndef ETEST_CORE_COMMON_GLOBAL_EXCEPTION_HANDLER_H_
#define ETEST_CORE_COMMON_GLOBAL_EXCEPTION_HANDLER_H_

#include <QObject>
#include <QMutex>
#include <QThread>
#include <csignal>

namespace etest {
namespace core {
namespace common {

class GlobalExceptionHandler : public QObject {
  Q_OBJECT

 public:
  static GlobalExceptionHandler& instance();

  void init();
  void shutdown();

 signals:
  void exceptionCaught(const QString& message, const QString& stackTrace);

 private:
  GlobalExceptionHandler();
  ~GlobalExceptionHandler() override;

  static void signalHandler(int signal);
  static void qtMessageHandler(QtMsgType type, const QMessageLogContext& context,
                     const QString& message);

  void setupSignalHandlers();
  void restoreSignalHandlers();

  static QList<int> s_signals_;
  static QList<signal_handler_t> s_oldHandlers_;
  bool initialized_ = false;
};

}  // namespace common
}  // namespace core
}  // namespace etest

#endif  // ETEST_CORE_COMMON_GLOBAL_EXCEPTION_HANDLER_H_
```

```cpp
// src/core/common/GlobalExceptionHandler.cpp
#include "GlobalExceptionHandler.h"
#include "Logger.h"

#ifdef Q_OS_WIN
#include <windows.h>
#endif

using namespace etest::core::logger;

namespace etest {
namespace core {
namespace common {

QList<int> GlobalExceptionHandler::s_signals_ = {
#ifdef Q_OS_WIN
  SIGABRT, SIGFPE, SIGILL, SIGSEGV, SIGINT, SIGTERM
#else
  SIGABRT, SIGFPE, SIGILL, SIGSEGV, SIGBUS, SIGINT, SIGTERM
#endif
};

QList<signal_handler_t> GlobalExceptionHandler::s_oldHandlers_;

GlobalExceptionHandler& GlobalExceptionHandler::instance() {
  static GlobalExceptionHandler handler;
  return handler;
}

GlobalExceptionHandler::GlobalExceptionHandler() : QObject(nullptr) {
}

GlobalExceptionHandler::~GlobalExceptionHandler() {
  if (initialized_) {
    shutdown();
  }
}

void GlobalExceptionHandler::init() {
  if (initialized_) return;

  // 安装Qt消息处理器
  qInstallMessageHandler(qtMessageHandler);

  // 安装信号处理器
  setupSignalHandlers();

  initialized_ = true;
  LOG_INFO("EXCEPTION", "全局异常捕获已初始化");
}

void GlobalExceptionHandler::shutdown() {
  if (!initialized_) return;

  // 恢复信号处理器
  restoreSignalHandlers();

  // 恢复Qt消息处理器
  qInstallMessageHandler(nullptr);

  initialized_ = false;
  LOG_INFO("EXCEPTION", "全局异常捕获已关闭");
}

void GlobalExceptionHandler::setupSignalHandlers() {
#ifdef Q_OS_WIN
  // Windows使用signal()，仅支持有限的信号类型
  for (int sig : s_signals_) {
    signal(sig, signalHandler);
  }
#else
  // Linux使用sigaction()，支持更多信号
  struct sigaction sa;
  sa.sa_handler = signalHandler;
  sigemptyset(&sa.sa_mask);
  sa.sa_flags = SA_RESTART;

  for (int sig : s_signals_) {
    sigaction(sig, &sa, nullptr);
  }
#endif
}

void GlobalExceptionHandler::restoreSignalHandlers() {
  for (int i = 0; i < s_signals_.size(); ++i) {
    signal(s_signals_[i], s_oldHandlers_[i]);
  }
  s_oldHandlers_.clear();
}

void GlobalExceptionHandler::signalHandler(int signal) {
  QString sigName;
  switch (signal) {
    case SIGABRT: sigName = "SIGABRT"; break;
    case SIGFPE:  sigName = "SIGFPE"; break;
    case SIGILL:  sigName = "SIGILL"; break;
    case SIGSEGV: sigName = "SIGSEGV"; break;
    case SIGBUS:  sigName = "SIGBUS"; break;
    case SIGINT:  sigName = "SIGINT"; break;
    case SIGTERM: sigName = "SIGTERM"; break;
    default:    sigName = QString("Signal %1").arg(signal); break;
  }

  LOG_FATAL("EXCEPTION", "收到信号 {}，程序即将退出", sigName);

  // 获取调用栈
#ifdef Q_OS_WIN
  // Windows下可以获取更详细的堆栈信息
  void* buffer[64];
  int count = CaptureStackBackTrace(0, 64, buffer, nullptr);
  QString stackTrace = "Stack trace:\n";
  for (int i = 0; i < count; ++i) {
    stackTrace += QString("  #%1: %2\n").arg(i).arg((quint64)buffer[i]);
  }
  LOG_FATAL("EXCEPTION", "调用栈:\n{}", stackTrace);
#endif

  // 立即刷新日志
  Logger::shutdown();

  // 恢复默认信号处理并重新发送信号
  signal(signal, SIG_DFL);
  raise(signal);
}

void GlobalExceptionHandler::qtMessageHandler(QtMsgType type,
                                          const QMessageLogContext& context,
                                          const QString& message) {
  // 输出到日志系统
  switch (type) {
    case QtDebugMsg:
      LOG_DEBUG("QT", message);
      break;
    case QtInfoMsg:
      LOG_INFO("QT", message);
      break;
    case QtWarningMsg:
      LOG_WARN("QT", message);
      break;
    case QtCriticalMsg:
      LOG_ERROR("QT", message);
      break;
    case QtFatalMsg:
      LOG_FATAL("QT", "致命错误: {}", message);
      // 输出文件位置
      if (context.file) {
        LOG_FATAL("QT", "位置: {}:{}", context.file, context.line);
      }
      // Qt致命错误默认会abort
      Logger::shutdown();
      exit(1);
      break;
  }
}

}  // namespace common
}  // namespace core
}  // namespace etest
```

## 六、模块初始化顺序

```cpp
// main.cpp 正确初始化顺序
#include "Logger.h"
#include "CrashHandler.h"
#include "GlobalExceptionHandler.h"
#include "ConfigManager.h"

int main(int argc, char* argv[]) {
  QApplication app(argc, argv);

  // 1. 初始化日志系统（最先）
  etest::core::logger::Logger::init();

  // 2. 初始化全局异常捕获
  etest::core::common::GlobalExceptionHandler::init();

  // 3. 崩溃捕获（在异常捕获后）
  etest::core::crashhandler::CrashHandler::init();

  // 4. 恢复窗口状态
  etest::core::config::ConfigManager::instance().restoreWindowState();

  // 5. 初始化主窗口
  MainWindow window;
  window.show();

  // 6. 应用退出时
  int result = app.exec();

  // 7. 退出前保存状态
  etest::core::config::ConfigManager::instance().saveWindowState();

  // 8. 关闭崩溃捕获
  etest::core::crashhandler::CrashHandler::shutdown();

  // 9. 关闭异常捕获
  etest::core::common::GlobalExceptionHandler::shutdown();

  // 10. 关闭日志系统（最后）
  etest::core::logger::Logger::shutdown();

  return result;
}
```

## 八、验证方法

### 日志系统验证

1. 程序启动后，日志输出到控制台和文件
2. 修改`log/level`配置，日志输出级别相应变化
3. 检查`%DOCUMENT%/etest/logs/`目录有日志文件

### 崩溃捕获验证

1. 制造崩溃（如`int* p = nullptr; *p = 1;`）
2. 程序生成崩溃日志文件到`%DOCUMENT%/etest/crash/`

### 配置管理验证

1. 修改配置后关闭程序，配置持久化到INI文件
2. 重新打开程序，配置值已恢复
3. 调用`exportToJson()`导出配置，使用`importFromJson()`可导入

### 工具类验证

1. FileUtil：创建/读取/删除文件正常
2. TimeUtil：时间格式化正确
3. StringUtil：编码转换正确（UTF-8/GBK）
4. ByteUtil：CRC校验结果正确

### 异常框架验证

1. 主线程抛出未捕获异常，程序记录日志并退出
2. 工作线程抛出未捕获异常，不影响主线程
3. Qt致命消息（如`qFatal()`）被捕获并记录
