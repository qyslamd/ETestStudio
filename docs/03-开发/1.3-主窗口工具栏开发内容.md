### 1.2 MainWindow.cpp

```cpp
void MainWindow::createToolBar() {
  // ==================== 文件工具栏 ====================
  file_toolbar_ = addToolBar(QStringLiteral("文件"));
  file_toolbar_->setObjectName("FileToolbar");
  file_toolbar_->setMovable(false);  // 固定在顶部

  toolbar_new_project_action_ = file_toolbar_->addAction(
      QIcon(":/icons/new.png"), QStringLiteral("新建项目"), this, &MainWindow::onNewProject);
  toolbar_new_project_action_->setToolTip(QStringLiteral("新建项目 (Ctrl+Shift+N)"));

  toolbar_open_project_action_ = file_toolbar_->addAction(
      QIcon(":/icons/open.png"), QStringLiteral("打开项目"), this, &MainWindow::onOpenProject);
  toolbar_open_project_action_->setToolTip(QStringLiteral("打开项目 (Ctrl+O)"));

  file_toolbar_->addSeparator();

  toolbar_save_action_ = file_toolbar_->addAction(
      QIcon(":/icons/save.png"), QStringLiteral("保存"), this, &MainWindow::onSaveFile);
  toolbar_save_action_->setToolTip(QStringLiteral("保存 (Ctrl+S)"));
  toolbar_save_action_->setEnabled(false);  // 初始禁用

  // ==================== 编辑工具栏 ====================
  edit_toolbar_ = addToolBar(QStringLiteral("编辑"));
  edit_toolbar_->setObjectName("EditToolbar");
  edit_toolbar_->setMovable(false);

  toolbar_undo_action_ = edit_toolbar_->addAction(
      QIcon(":/icons/undo.png"), QStringLiteral("撤销"), this, &MainWindow::onUndo);
  toolbar_undo_action_->setToolTip(QStringLiteral("撤销 (Ctrl+Z)"));
  toolbar_undo_action_->setEnabled(false);

  toolbar_redo_action_ = edit_toolbar_->addAction(
      QIcon(":/icons/redo.png"), QStringLiteral("重做"), this, &MainWindow::onRedo);
  toolbar_redo_action_->setToolTip(QStringLiteral("重做 (Ctrl+Y)"));
  toolbar_redo_action_->setEnabled(false);

  edit_toolbar_->addSeparator();

  toolbar_cut_action_ = edit_toolbar_->addAction(
      QIcon(":/icons/cut.png"), QStringLiteral("剪切"), this, &MainWindow::onCut);
  toolbar_cut_action_->setToolTip(QStringLiteral("剪切 (Ctrl+X)"));
  toolbar_cut_action_->setEnabled(false);

  toolbar_copy_action_ = edit_toolbar_->addAction(
      QIcon(":/icons/copy.png"), QStringLiteral("复制"), this, &MainWindow::onCopy);
  toolbar_copy_action_->setToolTip(QStringLiteral("复制 (Ctrl+C)"));
  toolbar_copy_action_->setEnabled(false);

  toolbar_paste_action_ = edit_toolbar_->addAction(
      QIcon(":/icons/paste.png"), QStringLiteral("粘贴"), this, &MainWindow::onPaste);
  toolbar_paste_action_->setToolTip(QStringLiteral("粘贴 (Ctrl+V)"));
  toolbar_paste_action_->setEnabled(false);
}
```

### 2.2 initSignals() 修改

添加工具栏按钮状态更新逻辑：

```cpp
void MainWindow::initSignals() {
  // ... 现有代码

  // 编辑器：更新工具栏按钮状态
  connect(editor_manager_, &EditorManager::currentEditorChanged, this,
          [this](EditorWidget* editor) {
            bool hasEditor = (editor != nullptr);
            bool hasSelection = hasEditor && editor->editor()->hasSelectedText();

            toolbar_save_action_->setEnabled(hasEditor && editor->isModified());
            toolbar_cut_action_->setEnabled(hasSelection);
            toolbar_copy_action_->setEnabled(hasSelection);
            toolbar_undo_action_->setEnabled(hasEditor && editor->editor()->canUndo());
            toolbar_redo_action_->setEnabled(hasEditor && editor->editor()->canRedo());
          });
}
```

```cpp
void MainWindow::onUndo() {
  if (auto* editor = editor_manager_->currentEditor()) {
    editor->editor()->undo();
  }
}

void MainWindow::onRedo() {
  if (auto* editor = editor_manager_->currentEditor()) {
    editor->editor()->redo();
  }
}

  }

void MainWindow::onCopy() {

    editor->editor()->copy();
  }
}

void MainWindow::onPaste() {

  }
}
```

| `:/icons/copy.png` | 复制 |
| `:/icons/paste.png` | 粘贴 |

### 3.2 添加方式

**方案A**：使用Qt内置图标
```cpp
toolbar_new_project_action_ = file_toolbar_->addAction(
    QStyle::standardIcon(QStyle::SP_DialogOpenButton),
    QStringLiteral("新建项目"), this, &MainWindow::onNewProject);
```

**方案B**：使用自定义资源（推荐）
```cpp
// resources.qrc
```

### 4.2 保存/恢复工具栏状态

```cpp
void MainWindow::closeEvent(QCloseEvent* event) {
  // 保存工具栏可见性
  ConfigManager::instance().set(CONFIG_TOOLBAR_VISIBLE,
                          file_toolbar_->isVisible());

  // ... 现有保存逻辑
  QMainWindow::closeEvent(event);
}

void MainWindow::restoreWindowState() {
  // ... 现有恢复逻辑

  // 恢复工具栏可见性
  bool toolbarVisible = ConfigManager::instance().get<bool>(
      CONFIG_TOOLBAR_VISIBLE, CONFIG_TOOLBAR_DEFAULT_VISIBLE);
  file_toolbar_->setVisible(toolbarVisible);
  edit_toolbar_->setVisible(toolbarVisible);
}
```

## 六、验证方法

1. **工具栏显示验证**：
   - 程序启动后，工具栏显示在顶部
   - 包含指定按钮（新建/打开/保存/撤销/重做/剪切/复制/粘贴）

2. **按钮功能验证**：
   - 点击"新建项目"弹出新建项目对话框
   - 点击"打开项目"弹出打开项目对话框
   - 打开文件后，"保存"按钮启用
   - 编辑内容后，"撤销"/"重做"按钮启用
   - 有选中文本时，"剪切"/"复制"按钮启用

3. **状态同步验证**：
   - 切换编辑器tab，工具栏按钮状态同步更新
   - 关闭所有文件后，按钮恢复禁用状态

4. **工具栏配置验证**：
   - 隐藏工具栏后关闭程序
   - 重新打开程序，工具栏保持隐藏状态

5. **整体验证**：
   - 工具栏可拖拽调整位置（默认固定）
   - 右键菜单可显示/隐藏工具栏
   - 右侧属性面板保持占位
