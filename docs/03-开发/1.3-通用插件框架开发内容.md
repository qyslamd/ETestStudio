## 二、核心类设计

### 2.1 PluginEventBus（插件事件总线）

基于Qt全局信号实现插件间通信，类似事件总线模式。

```cpp
// src/core/plugin/PluginEventBus.h
#ifndef ETEST_CORE_PLUGIN_PLUGIN_EVENT_BUS_H_
#define ETEST_CORE_PLUGIN_PLUGIN_EVENT_BUS_H_

#include <QObject>
#include <QVariantMap>

namespace etest {
namespace core {
namespace plugin {

// 事件类型枚举
enum class EventType {
    PluginLoaded,        // 插件加载完成
    PluginUnloaded,      // 插件卸载完成
    ProjectOpened,       // 项目打开
    ProjectClosed,       // 项目关闭
    FileOpened,          // 文件打开
    FileSaved,           // 文件保存
    ConfigChanged,       // 配置变更
    Custom              // 自定义事件
};

// 事件对象
struct PluginEvent {
    QString senderId;       // 发送者插件ID
    EventType type;         // 事件类型
    QString topic;          // 事件主题（用于Custom类型）
    QVariantMap data;      // 事件数据
    QDateTime timestamp;   // 时间戳
};

class PluginEventBus : public QObject {
    Q_OBJECT

 public:
    static PluginEventBus& instance();

    // 订阅事件
    void subscribe(const QString& pluginId, const QString& topic,
                 std::function<void(const PluginEvent&)> callback);
    void subscribe(const QString& pluginId, EventType type,
                 std::function<void(const PluginEvent&)> callback);

    // 取消订阅
    void unsubscribe(const QString& pluginId);
    void unsubscribe(const QString& pluginId, const QString& topic);
    void unsubscribe(const QString& pluginId, EventType type);

    // 发布事件
    void publish(const QString& senderId, EventType type,
              const QVariantMap& data = QVariantMap());
    void publish(const QString& senderId, const QString& topic,
              const QVariantMap& data = QVariantMap());

 signals:
    void eventReceived(const PluginEvent& event);

 private:
    PluginEventBus();
    ~PluginEventBus() override;

    struct Subscriber {
        QString pluginId;
        QString topic;  // 空字符串表示订阅所有主题
        EventType type;  // 仅当topic为空时有效
        std::function<void(const PluginEvent&)> callback;
    };

    QList<Subscriber> subscribers_;
};

}  // namespace plugin
}  // namespace core
}  // namespace etest

#endif  // ETEST_CORE_PLUGIN_PLUGIN_EVENT_BUS_H_
```

**使用示例**：
```cpp
// 插件A订阅ProjectOpened事件
PluginEventBus::instance().subscribe("plugin_a", EventType::ProjectOpened,
    [](const PluginEvent& event) {
        QString projectPath = event.data.value("path").toString();
        qDebug() << "插件A收到项目打开事件:" << projectPath;
    });

// 插件B发布ProjectOpened事件
PluginEventBus::instance().publish("plugin_b", EventType::ProjectOpened,
    {{"path", "/work/myproject"}});
```

### 2.2 PluginConfig（插件配置基类）

每个插件独立的INI配置文件。

```cpp
// src/core/plugin/PluginConfig.h
#ifndef ETEST_CORE_PLUGIN_PLUGIN_CONFIG_H_
#define ETEST_CORE_PLUGIN_PLUGIN_CONFIG_H_

#include <QObject>
#include <QString>
#include <QVariant>

namespace etest {
namespace core {
namespace plugin {

class PluginConfig : public QObject {
    Q_OBJECT

 public:
    explicit PluginConfig(const QString& pluginId, QObject* parent = nullptr);
    ~PluginConfig() override;

    // 读取配置
    QVariant value(const QString& key, const QVariant& defaultValue = QVariant()) const;
    template<typename T>
    T value(const QString& key, const T& defaultValue) const {
        return value(key, QVariant::fromValue(defaultValue)).template value<T>();
    }

    // 写入配置
    void setValue(const QString& key, const QVariant& value);

    // 保存到文件
    bool save();

    // 重新加载
    bool reload();

    // 重置为默认值
    void reset();

    // 检查键是否存在
    bool contains(const QString& key) const;

    // 获取配置文件的路径
    QString filePath() const;

 signals:
    void valueChanged(const QString& key, const QVariant& value);

 private:
    QString plugin_id_;
    QString file_path_;
    QVariantMap values_;
};

}  // namespace plugin
}  // namespace core
}  // namespace etest

#endif  // ETEST_CORE_PLUGIN_PLUGIN_CONFIG_H_
```

**配置文件位置**：
```
%APPDATA%/etest/plugins/{plugin_id}/config.ini
```

例如：
```
C:\Users\username\AppData\Roaming\etest\plugins\hello_plugin\config.ini
```

**INI格式示例**：
```ini
[Settings]
auto_connect=true
port=COM3
baudrate=115200

[UI]
window_x=100
window_y=100
window_width=800
window_height=600
```

**使用示例**：
```cpp
// MyPlugin类中
class MyPlugin : public IPlugin {
 public:
    bool initialize() override {
        config_ = new PluginConfig("my.plugin.id", this);
        config_->setValue("auto_connect", true);
        config_->save();
        return true;
    }

 private:
    PluginConfig* config_ = nullptr;
};
```

### 2.3 Hello Plugin（示例插件）

演示通信和持久化的示例插件。

```
examples/plugins/hello_plugin/
├── CMakeLists.txt
├── hello_plugin.h
├── hello_plugin.cpp
├── config.json    # 插件元数据
└── resources/
    └── icon.png   # 插件图标（可选）
```

```cpp
// examples/plugins/hello_plugin/hello_plugin.h
#include <IPlugin>

namespace etest {
namespace example {

class HelloPlugin : public QObject, public IPlugin {
    Q_OBJECT
    Q_INTERFACES(etest::core::plugin::IPlugin)

 public:
    explicit HelloPlugin(QObject* parent = nullptr);
    ~HelloPlugin() override;

    // IPlugin接口
    bool initialize() override;
    bool start() override;
    void stop() override;
    void uninitialize() override;

    PluginMetaData metaData() const override;
    bool isRunning() const override;

 private:
    void onProjectOpened(const PluginEvent& event);
    void onConfigChanged(const PluginEvent& event);

    PluginMetaData meta_data_;
    PluginConfig* config_ = nullptr;
    bool running_ = false;
};

}  // namespace example
}  // namespace etest
#endif  // ETEST_EXAMPLE_HELLO_PLUGIN_H_
```

```cpp
// examples/plugins/hello_plugin/hello_plugin.cpp
#include "hello_plugin.h"

namespace etest {
namespace example {

HelloPlugin::HelloPlugin(QObject* parent)
    : QObject(parent),
      meta_data_({
          "example.hello",    // ID
          "Hello插件示例",   // 名称
          "1.0.0",         // 版本
          "演示插件间通信和配置持久化的示例插件",  // 描述
          "Example",        // 作者
          "example",       // 分类
          {}              // 依赖
      }) {}

bool HelloPlugin::initialize() {
    // 创建配置
    config_ = new PluginConfig("example.hello", this);

    // 订阅事件
    PluginEventBus::instance().subscribe(
        "example.hello", EventType::ProjectOpened,
        [this](const PluginEvent& e) { onProjectOpened(e); });

    PluginEventBus::instance().subscribe(
        "example.hello", EventType::ConfigChanged,
        [this](const PluginEvent& e) { onConfigChanged(e); });

    // 加载配置
    config_->reload();
    LOG_INFO("HELLO_PLUGIN", "插件初始化完成");
    return true;
}

bool HelloPlugin::start() {
    running_ = true;
    LOG_INFO("HELLO_PLUGIN", "插件启动");
    return true;
}

void HelloPlugin::stop() {
    running_ = false;
    LOG_INFO("HELLO_PLUGIN", "插件停止");
}

void HelloPlugin::uninitialize() {
    // 保存配置
    if (config_) {
        config_->save();
    }

    // 取消订阅
    PluginEventBus::instance().unsubscribe("example.hello");
    LOG_INFO("HELLO_PLUGIN", "插件卸载");
}

PluginMetaData HelloPlugin::metaData() const {
    return meta_data_;
}

bool HelloPlugin::isRunning() const {
    return running_;
}

void HelloPlugin::onProjectOpened(const PluginEvent& event) {
    QString path = event.data.value("path").toString();
    LOG_INFO("HELLO_PLUGIN", "收到项目打开事件: {}", path);

    // 可以发布自定义事件
    PluginEventBus::instance().publish(
        "example.hello", EventType::Custom,
        {{"action", "project_loaded"}, {"path", path}});
}

void HelloPlugin::onConfigChanged(const PluginEvent& event) {
    QString key = event.data.value("key").toString();
    QVariant value = event.data.value("value");
    LOG_INFO("HELLO_PLUGIN", "配置变更: {} = {}", key, value);
}

}  // namespace example
}  // namespace etest

// 插件导出宏
#define HelloPlugin_iid "etest.example.hello/1.0"
Q_PLUGIN_METADATA(IID HelloPlugin_iid FILE "config.json")
```

**config.json**：
```json

    "id": "example.hello",
    "name": "Hello插件示例",
    "version": "1.0.0",
    "description": "演示插件间通信和配置持久化的示例插件",
    "author": "Example",
    "category": "example",
    "dependencies": []
}
```

## 五、验证方法

1. **事件总线验证**：
   - 加载hello_plugin后，打开项目，控制台输出"收到项目打开事件"
   - 证明事件总线通信正常

2. **配置持久化验证**：
   - 修改配置后关闭程序，检查`%APPDATA%/etest/plugins/hello_plugin/config.ini`文件存在
   - 重新打开程序，配置值已持久化

3. **示例插件验证**：
   - 程序启动时加载hello_plugin
   - 输出日志显示插件生命周期（initialize/start/stop/uninitialize）
   - 程序正常退出，无崩溃

4. **信号订阅验证**：
   - 订阅其他插件发布的事件能正常接收
   - 取消订阅后不再接收事件
